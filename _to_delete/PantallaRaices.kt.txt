private fun etiquetaAccion(a: String) = when (a) {
    "LOGIN" -> "Entrada"
    "USUARIO" -> "Cuentas"
    "INVITADO" -> "Visitas"
    "RECUPERAR" -> "Recuperación"
    "SESION" -> "Sesión"
    else -> a
}

private fun plural(n: Int, uno: String, varios: String) = "$n ${if (n == 1) uno else varios}"

@Composable
fun PantallaRaices(c: Contenedor, yo: Usuario) {
    val scope = rememberCoroutineScope()
    val snack = remember { SnackbarHostState() }
    var cuentas by remember { mutableStateOf<List<CuentaInfo>>(emptyList()) }
    var pases by remember { mutableStateOf<List<Invitacion>>(emptyList()) }
    var movimientos by remember { mutableStateOf<List<RegistroAuditoria>>(emptyList()) }
    var recargar by remember { mutableIntStateOf(0) }
    var claveTemporal by remember { mutableStateOf<Pair<String, String>?>(null) }   // cuenta, clave
    var porEliminar by remember { mutableStateOf<CuentaInfo?>(null) }

    fun aviso(t: String) { scope.launch { snack.showSnackbar(t) } }

    LaunchedEffect(recargar) {
        val datos = withContext(Dispatchers.IO) {
            Triple(c.usuarios.listarDetalle(), c.invitaciones.activas(), c.db.auditoriaDeAcceso(20))
        }
        cuentas = datos.first
        pases = datos.second
        movimientos = datos.third
    }

    /** Ejecuta una acción de administración, la audita y refresca el panel. */
    fun accion(detalle: String, bloque: () -> String?) {
        scope.launch {
            val e = withContext(Dispatchers.IO) {
                bloque().also { c.db.auditar(yo.nombre, "USUARIO", detalle, if (it == null) "OK" else "FALLA") }
            }
            aviso(e ?: "Listo: ${detalle.replaceFirstChar { it.lowercase() }}")
            recargar++
        }
    }

    Scaffold(snackbarHost = { SnackbarHost(snack) }, containerColor = MaterialTheme.colorScheme.background) { relleno ->
        LazyColumn(
            Modifier.fillMaxSize().padding(relleno),
            contentPadding = PaddingValues(16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            item {
                Column(Modifier.padding(top = 8.dp, bottom = 4.dp), verticalArrangement = Arrangement.spacedBy(4.dp)) {
                    Text("Raíces del invernadero", style = MaterialTheme.typography.headlineSmall, modifier = Modifier.semantics { heading() })
                    Text(
                        "Quién cuida GreenBox, quién está de visita y qué ha pasado en la puerta.",
                        style = MaterialTheme.typography.bodyMedium,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                    val bloqueadas = cuentas.count { it.bloqueada }
                    Text(
                        listOfNotNull(
                            plural(cuentas.size, "cuenta", "cuentas"),
                            plural(cuentas.count { it.usuario.rol == Rol.ADMIN }, "administrador", "administradores"),
                            plural(pases.size, "pase abierto", "pases abiertos"),
                            if (bloqueadas > 0) plural(bloqueadas, "bloqueada", "bloqueadas") else null
                        ).joinToString(" · "),
                        style = MaterialTheme.typography.bodyMedium,
                        color = MaterialTheme.colorScheme.secondary
                    )
                }
            }

            item {
                Seccion("Cuentas") {
                    cuentas.forEachIndexed { i, cuenta ->
                        if (i > 0) HorizontalDivider()
                        FilaCuenta(
                            cuenta = cuenta,
                            soyYo = cuenta.usuario.id == yo.id,
                            alCambiarRol = { rol -> accion("Cambiar ${cuenta.usuario.nombre} a ${rol.etiqueta}") { c.usuarios.cambiarRol(cuenta.usuario.id, rol) } },
                            alDesbloquear = { accion("Desbloquear ${cuenta.usuario.nombre}") { c.usuarios.desbloquear(cuenta.usuario.id); null } },
                            alRestablecer = {
                                scope.launch {
                                    val clave = withContext(Dispatchers.Default) { c.usuarios.restablecerPorAdmin(cuenta.usuario.id) }
                                    withContext(Dispatchers.IO) { c.db.auditar(yo.nombre, "USUARIO", "Contraseña temporal para ${cuenta.usuario.nombre}", "OK") }
                                    claveTemporal = cuenta.usuario.nombre to clave
                                    recargar++
                                }
                            },
                            alEliminar = { porEliminar = cuenta }
                        )
                    }
                    HorizontalDivider()
                    NuevaCuenta(c, yo) { aviso(it); recargar++ }
                }
            }

            item {
                Seccion("Pases de visita abiertos") {
                    if (pases.isEmpty()) {
                        Text("Nadie tiene la puerta abierta ahora.", style = MaterialTheme.typography.bodyMedium, color = MaterialTheme.colorScheme.onSurfaceVariant)
                    }
                    pases.forEach { p ->
                        FilaPase(p, mostrarCreador = true) {
                            scope.launch {
                                withContext(Dispatchers.IO) {
                                    c.invitaciones.revocar(p.id, yo)
                                    c.db.auditar(yo.nombre, "INVITADO", "Revocar pase #${p.id} de ${p.creador}", "OK")
                                }
                                aviso("Pase revocado")
                                recargar++
                            }
                        }
                    }
                }
            }

            item {
                Seccion("Movimiento reciente") {
                    if (movimientos.isEmpty()) {
                        Text("Todavía no hay movimiento.", style = MaterialTheme.typography.bodyMedium, color = MaterialTheme.colorScheme.onSurfaceVariant)
                    }
                    movimientos.forEach { m -> FilaMovimiento(m) }
                }
            }
        }
    }

    claveTemporal?.let { (cuenta, clave) ->
        val contexto = LocalContext.current
        AlertDialog(
            onDismissRequest = {},
            title = { Text("Contraseña temporal") },
            text = {
                Column(verticalArrangement = Arrangement.spacedBy(12.dp)) {
                    Text("Entrégasela a $cuenta en persona. Con ella entra y luego puede generar su semilla en Ajustes. No volverás a verla.")
                    SelectionContainer {
                        Text(clave, style = MaterialTheme.typography.titleLarge.copy(fontFamily = FontFamily.Monospace, letterSpacing = 2.sp))
                    }
                }
            },
            confirmButton = { TextButton(onClick = { claveTemporal = null }) { Text("Listo") } },
            dismissButton = { TextButton(onClick = { copiarSensible(contexto, "Contraseña temporal", clave) }) { Text("Copiar") } }
        )
    }

    porEliminar?.let { cuenta ->
        AlertDialog(
            onDismissRequest = { porEliminar = null },
            title = { Text("¿Eliminar a ${cuenta.usuario.nombre}?") },
            text = { Text("La cuenta deja de existir en este invernadero. Su historial de auditoría se conserva.") },
            confirmButton = {
                TextButton(onClick = {
                    porEliminar = null
                    accion("Eliminar ${cuenta.usuario.nombre}") { c.usuarios.eliminar(cuenta.usuario.id) }
                }) { Text("Eliminar", color = MaterialTheme.colorScheme.error) }
            },
            dismissButton = { TextButton(onClick = { porEliminar = null }) { Text("Cancelar") } }
        )
    }
}

@Composable
private fun FilaCuenta(
    cuenta: CuentaInfo,
    soyYo: Boolean,
    alCambiarRol: (Rol) -> Unit,
    alDesbloquear: () -> Unit,
    alRestablecer: () -> Unit,
    alEliminar: () -> Unit
) {
    var menu by remember { mutableStateOf(false) }
    Row(Modifier.fillMaxWidth().heightIn(min = 56.dp), verticalAlignment = Alignment.CenterVertically) {
        Column(Modifier.weight(1f)) {
            Text(cuenta.usuario.nombre + if (soyYo) " (tú)" else "", style = MaterialTheme.typography.bodyLarge)
            Row(horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                Text(cuenta.usuario.idVisible.orEmpty(), style = MaterialTheme.typography.bodySmall, color = MaterialTheme.colorScheme.onSurfaceVariant)
                Text("· " + cuenta.usuario.rol.etiqueta, style = MaterialTheme.typography.bodySmall, color = MaterialTheme.colorScheme.secondary)
                if (cuenta.bloqueada) Text("· bloqueada", style = MaterialTheme.typography.bodySmall, color = MaterialTheme.colorScheme.error)
                if (!cuenta.tieneSemilla) Text("· sin semilla", style = MaterialTheme.typography.bodySmall, color = MaterialTheme.colorScheme.onSurfaceVariant)
            }
        }
        if (!soyYo) {
            Box {
                IconButton(onClick = { menu = true }) { Icon(Icons.Filled.MoreVert, contentDescription = "Opciones de ${cuenta.usuario.nombre}") }
                DropdownMenu(expanded = menu, onDismissRequest = { menu = false }) {
                    Rol.entries.filter { it != cuenta.usuario.rol }.forEach { r ->
                        DropdownMenuItem(text = { Text("Hacer ${r.etiqueta}") }, onClick = { menu = false; alCambiarRol(r) })
                    }
                    if (cuenta.bloqueada) DropdownMenuItem(text = { Text("Desbloquear") }, onClick = { menu = false; alDesbloquear() })
                    DropdownMenuItem(text = { Text("Restablecer contraseña") }, onClick = { menu = false; alRestablecer() })
                    DropdownMenuItem(
                        text = { Text("Eliminar cuenta", color = MaterialTheme.colorScheme.error) },
                        onClick = { menu = false; alEliminar() }
                    )
                }
            }
        }
    }
}

@Composable
private fun FilaMovimiento(m: RegistroAuditoria) {
    val falla = m.resultado != "OK"
    Column(Modifier.fillMaxWidth().padding(vertical = 4.dp)) {
        Text("${etiquetaAccion(m.accion)} · ${m.usuario}", style = MaterialTheme.typography.bodyMedium)
        Text(
            "${formatoFecha(m.ts)} · ${m.detalle}" + if (falla) " · ${m.resultado}" else "",
            style = MaterialTheme.typography.bodySmall,
            color = if (falla) MaterialTheme.colorScheme.error else MaterialTheme.colorScheme.onSurfaceVariant
        )
    }
}

@Composable
private fun NuevaCuenta(c: Contenedor, yo: Usuario, aviso: (String) -> Unit) {
    val scope = rememberCoroutineScope()
    var abierta by rememberSaveable { mutableStateOf(false) }
    var nombre by remember { mutableStateOf("") }
    var clave by remember { mutableStateOf("") }
    var rol by remember { mutableStateOf(Rol.OPERADOR) }
    var guardando by remember { mutableStateOf(false) }

    if (!abierta) {
        OutlinedButton(onClick = { abierta = true }, modifier = Modifier.fillMaxWidth()) { Text("Plantar una cuenta nueva") }
        return
    }
    Text("Cuenta nueva", style = MaterialTheme.typography.titleSmall)
    OutlinedTextField(
        nombre, { nombre = it.take(30) }, label = { Text("Nombre") }, singleLine = true,
        keyboardOptions = KeyboardOptions(autoCorrectEnabled = false), modifier = Modifier.fillMaxWidth()
    )
    CampoClave(clave, "Contraseña", { clave = it.take(64) }, ayuda = "Mínimo 8 caracteres, con letras y números")
    Column(Modifier.selectableGroup()) {
        Rol.entries.forEach { r -> OpcionRadio(r.etiqueta, rol == r) { rol = r } }
    }
    Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
        TextButton(onClick = { abierta = false; nombre = ""; clave = "" }) { Text("Cancelar") }
        Button(
            onClick = {
                guardando = true
                val n = normalizarNombre(nombre)
                val arreglo = clave.toCharArray()
                val r = rol
                scope.launch {
                    val e = withContext(Dispatchers.Default) { c.usuarios.registrar(n, arreglo, r) }
                    arreglo.fill('\u0000')
                    withContext(Dispatchers.IO) { c.db.auditar(yo.nombre, "USUARIO", "Crear $n (${r.etiqueta})", if (e == null) "OK" else "FALLA") }
                    guardando = false
                    if (e == null) { nombre = ""; clave = ""; abierta = false; aviso("Cuenta de $n creada") } else aviso(e)
                }
            },
            enabled = !guardando && nombre.isNotBlank() && clave.isNotEmpty()
        ) { Text("Crear cuenta") }
    }
}
